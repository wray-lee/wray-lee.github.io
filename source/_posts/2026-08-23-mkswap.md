# mkswap

```bash
free -m

sudo dd if=/dev/zero of=/swap count=[space eg:4096] bs=1M

sudo chmod 600 /swap
sudo mkswap /swap
sudo swapon /swap


echo '/swap none swap sw 0 0' >> /etc/fstab


sudo dd if=/dev/zero of=/swap count=[space eg:4096] bs=1M && sudo chmod 600 /swap && sudo mkswap /swap && sudo swapon /swap 
```



# Rmswap

```bash
sudo swapoff -v /swap

# Remove last Line
sed -i '$d' /etc/fstab # Or `sed -i 's/swap none swap sw 0 0/g'`

sudo rm /swap
```



## Btrfs

```
# 1. 创建一个大小为 0 的空文件（以创建 4G 大小的 swap 为例）
sudo truncate -s 0 /swapfile

# 2. 禁用该文件的写时复制（+C 属性）
sudo chattr +C /swapfile

# 3. 禁用该文件的压缩属性
sudo btrfs property set /swapfile compression none

# 分配 4G 空间（可根据需要修改，如 8G 则为 8G）
sudo fallocate -l 4G /swapfile

# 设置正确的权限（仅 root 可读写，这是安全要求）
sudo chmod 600 /swapfile

# 格式化为 swap 格式
sudo mkswap /swapfile

# 启用 swap 文件
sudo swapon /swapfile

echo '/swapfile none swap defaults 0 0' | sudo tee -a /etc/fstab
```
