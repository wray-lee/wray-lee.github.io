# WSL

## 迁移

```bash
wsl --shutdown
wsl --unregister Ubuntu
wsl --export Arch D:/wsl/arch.tar
wsl --unregister Arch
wsl --import arch D:\wsl D:\wsl/arch.tar
```



## 压缩vhdx

### Fixed vhdx(unsparsed)

```bash 
# wsl中
sudo fstrim /
###
# windows中
diskpart
select vdisk file="D:\WSL-Arch\ext4.vhdx"
attach vdisk readonly
compact vdisk
detach vdisk

```

### wsl内剪贴板问题

```bash
curl -sLo /tmp/win32yank.zip https://github.com/equalsraf/win32yank/releases/latest/download/win32yank-x64.zip
unzip -p /tmp/win32yank.zip win32yank.exe > /tmp/win32yank.exe
chmod +x /tmp/win32yank.exe
sudo mv /tmp/win32yank.exe /usr/local/bin/
```



`.config/nvim/init.lua`

```lua
-- 开启系统剪贴板同步
vim.opt.clipboard = "unnamedplus"

-- 检测系统环境特征
local is_wsl = (vim.fn.has('wsl') == 1) or (os.getenv("WSL_DISTRO_NAME") ~= nil)
-- 检测是否处于 SSH 远程连接中
local is_ssh = (os.getenv("SSH_CLIENT") ~= nil) or (os.getenv("SSH_TTY") ~= nil)

if is_wsl and not is_ssh then
  -- 仅命中本地 WSL（非远程登录）：通过 Windows 宿主机的 exe 进行物理穿透
  vim.g.clipboard = {
    name = 'win32yank-wsl',
    copy = {
      ['+'] = 'win32yank.exe -i --crlf',
      ['*'] = 'win32yank.exe -i --crlf',
    },
    paste = {
      ['+'] = 'win32yank.exe -o --lf',
      ['*'] = 'win32yank.exe -o --lf',
    },
    cache_enabled = 0,
  }
else
  -- 命中 SSH 远程连接（含远程 WSL）或纯正 Linux：强制走 OSC 52 终端协议
  vim.g.clipboard = {
    name = 'OSC 52',
    copy = {
      ['+'] = require('vim.ui.clipboard.osc52').copy('+'),
      ['*'] = require('vim.ui.clipboard.osc52').copy('*'),
    },
    paste = {
      ['+'] = require('vim.ui.clipboard.osc52').paste('+'),
      ['*'] = require('vim.ui.clipboard.osc52').paste('*'),
    },
  }
end

-- 避免光标异常
vim.api.nvim_create_autocmd("VimLeave", {
    callback = function()
        -- \x1b 是 ESC 的标准十六进制表示，flush 确保序列立刻写入终端
        io.stdout:write("\x1b[5 q")
        io.stdout:flush()
    end
})

```

## 修复使用nvim后光标异常

`.zshrc`

```bash
precmd() {
    print -n '\e[5 q'
}
```



## 防火墙处理

内网穿透地址仍然受防火墙管控

```pwsh
Get-NetFirewallHyperVVMSetting

# 把所有 WSL 相关的 VM 默认入站改成允许
Get-NetFirewallHyperVVMSetting | ForEach-Object {
  Set-NetFirewallHyperVVMSetting -Name $_.Name -DefaultInboundAction Allow
}
```
