## fsモジュール

モジュールのインポート

```javascript
const fs = require('fs');
```

書き込み

```javascript
// 异步
fs.writeFile('./tmp', 'content', err => {
    if(err) {
        console.log('写入失败');
        return;
    }
    console.log("写入成功");
});

// 同步
fs.writeFileSync('./tmp', 'content');


//写入到脚本目录
fs.writeFile(__dirname + 'tmp', 'content', err =>{});
```

追加

```javascript
fs.appendFile('./tmp', 'content', err => {
    if(err){
        console.log('Failed')
        return;
    }
    console.log('success')
});

fs.appendFileSync('./tmp', 'content');

fs.writeFile('./tmp', 'content', {flag: a}, err => {
    if(err) {
        console.log('Failed');
        return;
    }
    console.log('success')
});
```

ストリーム書き込み

```javascript
const ws = fs.createWriteStream('./tmp');
ws.write('content\r\n');
ws.write('content\r\n');
ws.write('content');
ws.close();
```

読み取り

```javascript
//data为读取内容
fs.readFile('./tmp', (err, data) => {
    if(err) {
        console.log('Failed');
        return;
    }
    console.log('data.toString()')	//默认输出16进制hex编码
});

let data = fs.readFileSync('./tmp');
console.log(data);
```

ストリーム読み取り

```javascript
const rs = fs.createReadStream('./tmp');

rs.on('data', chunck => {
    console.log(chunk);
});

rs.on('end', () => {
    console.log('Finished');
});
```

パイプコピー

```javascript
rs.pipe(ws)
```

名前変更と移動

```javascript
//重命名
fs.rename('./tmp', './newname', err => {
    if(err) {
        console.log('');
        return;
    }
});

fs.rename('./tmp', './dir/tmp', err => {
    if(err) {
        console.log('');
        return;
    }
    console.log('')
});
```

削除

```javascript
// 将被移除
```

フォルダの作成

```javascript
fs.mkdir('./dir', err => {
    if(err) { console.log(''); return; }
    console.log('');
})

//递归创建
fs.mkdir('./a/b/c', {recursive: true}, err => {
    if(err) { console.log(''); return;}
    console.log('')
});
```

フォルダの読み取り

```javascript
fs.readdir('./dir', (err, data) => {
    if(err) { console.log(''); return; }
    console.log('')
});
```

情報の読み取り


## pathモジュール

`dir = __dirname + '/tmp' ` Windows で返されるディレクトリパスのスラッシュが統一されていない

resolve で結合することで解決できます

```javascript
path = require('path')
path.resolve(__dirname, './tmp')
path.resolve(绝对路径， 相对路径)
```
