## fs模块

导入模块

```javascript
const fs = require('fs');
```

写入

```javascript
// 异步
fs.writeFile('./tmp', 'content', err => {
    if(err) {
        console.log('写入失败');
        return;
    }
    console.log("写入成功");
});

// 同步
fs.writeFileSync('./tmp', 'content');


//写入到脚本目录
fs.writeFile(__dirname + 'tmp', 'content', err =>{});
```

追加

```javascript
fs.appendFile('./tmp', 'content', err => {
    if(err){
        console.log('Failed')
        return;
    }
    console.log('success')
});

fs.appendFileSync('./tmp', 'content');

fs.writeFile('./tmp', 'content', {flag: a}, err => {
    if(err) {
        console.log('Failed');
        return;
    }
    console.log('success')
});
```

流式写入

```javascript
const ws = fs.createWriteStream('./tmp');
ws.write('content\r\n');
ws.write('content\r\n');
ws.write('content');
ws.close();
```

读取

```javascript
//data为读取内容
fs.readFile('./tmp', (err, data) => {
    if(err) {
        console.log('Failed');
        return;
    }
    console.log('data.toString()')	//默认输出16进制hex编码
});

let data = fs.readFileSync('./tmp');
console.log(data);
```

流式读取

```javascript
const rs = fs.createReadStream('./tmp');

rs.on('data', chunck => {
    console.log(chunk);
});

rs.on('end', () => {
    console.log('Finished');
});
```

管道复制

```javascript
rs.pipe(ws)
```

重命名和移动

```javascript
//重命名
fs.rename('./tmp', './newname', err => {
    if(err) {
        console.log('');
        return;
    }
});

fs.rename('./tmp', './dir/tmp', err => {
    if(err) {
        console.log('');
        return;
    }
    console.log('')
});
```

删除

```javascript
// 将被移除
//fs.rmdir('./dir', err => {
//    if(err) { console.log();return; }
//    console.log
//})
fs.unlink('./tmp', err => {
    if(err) {
        console.log('');
        return;
    }
    console.log('');
})

fs.rm('./tmp', err => {
    if(err) {
        console.log('');
        return;
    }
    console.log('');
})
```

创建文件夹

```javascript
fs.mkdir('./dir', err => {
    if(err) { console.log(''); return; }
    console.log('');
})

//递归创建
fs.mkdir('./a/b/c', {recursive: true}, err => {
    if(err) { console.log(''); return;}
    console.log('')
});
```

读取文件夹

```javascript
fs.readdir('./dir', (err, data) => {
    if(err) { console.log(''); return; }
    console.log('')
});
```

读取信息

```javascript
fs.stat('./tmp', (err, data) => {
    if(err) { console.log(); return; }
    console.log()
});
//isFile 
```



## path模块

`dir = __dirname + '/tmp' ` windows下返回的目录路径斜杠不统一

可通过resolve拼接解决

```javascript
path = require('path')
path.resolve(__dirname, './tmp')
path.resolve(绝对路径， 相对路径)
```
