# Android包移植

使用unpacker解包卡刷包

## 原厂底包vendor

```bash
# 如果system/system/apex是文件

#vendor/build.prop
ro.apex.updatable=true

#############################################

# 如果system/system/apex是文件夹

#vendor/build.prop
ro.apex.updatable=true

```

> 不同厂
>
> vendor修改
>
> ```bash
> #vendor/etc
> 原厂包group和passwd复制到新的vendor/etc中
> ```
>





去除avb, data加密

```bash
#vendor/etc/fstab.*

# 删除
# /dev/block/by-name/userdata /data
fileencryption=*,keydirectory=/metadata/vold/metadata_encryption
# 或改成
encryptable=*,keydirectory=/metadata/vold/metadata_encryption
```



> 在镜像中关闭avb校验（fastboot可以替代）
>
> ```bash
> #vendor/etc/fstab.*
> 去除avb相关字样
> #avb
> #avb=
> #avb_keys=*
> #avb_keys=/avb/q-gsi.avbpubkey:/avb/r-gsi.avbpubkey:/avb/s-gsi.avbpubkey
> 
> ```
>
> 



内核宽容

```bash
#<vendor_boot>/<target>/BoardConfig.mk

androidboot.selinux=permissive #宽容
androidboot.selinux=enforcing  #严格
```

# 编译lineageOS

```bash
yay lineageos-devel
pacman -S repo
mkdir lineageos
cd lineageos
repo init -u https://github.com/LineageOS/android.git -b lineage-23.0 --git-lfs
repo sync -c -d

mkdir -p android/dump/system
mkdir -p android/dump/system/vendor
mkdir -p android/dump/system/vendor_dlkm
mkdir -p android/dump/system/product
mkdir -p android/dump/system/system_ext
mkdir -p android/dump/system/odm
# 复制解包镜像

cp system.img android/dump/system
cp vendor.img android/dump/system/vendor
cp vendor_dlkm.img android/dump/system/vendor_dlkm
cp product.img android/dump/system/product
cp system_ext.img android/dump/system/system_ext
cp odm.img android/dump/system/odm

# 设备树
mkdir -p device/{brand}/{code}
cp deviceTree/* device/{brand}/{code}
```
