# SMBマウント

```bash
apt install cifs-utils -y
mkdir -p /mnt/smb
mkdir -p /etc/samba

# /etc/fstab
//10.17.1.112/storage                    /mnt/smb        cifs    _netdev,credentials=/etc/samba/crt,iocharset=utf8,uid=www-data,gid=www-data 0 0
###

sudo mount -a
```
